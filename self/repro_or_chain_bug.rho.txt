fn s3(s: string) -> i32 {
  let mut i: usize = 0;
  let mut spaces: i32 = 0;
  while i < len(s) {
    let c: u8 = s[i];
    if c == 32 || c == 9 || c == 13 {
      spaces += 1;
      i += 1;
      continue;
    }
    i += 1;
  }
  return spaces;
}
