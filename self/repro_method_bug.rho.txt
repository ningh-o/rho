struct Vec[T] {
  items: []T,
  n: usize,
}
fn Vec.init(self: *Vec) {
  self.items = make([]T, 8);
  self.n = 0;
}
fn Vec.push(self: *Vec, v: T) {
  if self.n == len(self.items) {
    let grown: []T = make([]T, len(self.items) * 2);
    let mut i: usize = 0;
    while i < self.n {
      grown[i] = self.items[i];
      i += 1;
    }
    self.items = grown;
  }
  self.items[self.n] = v;
  self.n += 1;
}
fn Vec.last(self: *Vec) -> T {
  return self.items[self.n - 1];
}
